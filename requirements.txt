# ---------------------------------------------------------------------------
# Switchboard - runtime dependencies
#
# This is what the server needs to run, and what the Docker image installs.
# Evaluation, plotting and machine-learning packages live in
# requirements-dev.txt: they are large (PyTorch-adjacent, ~500MB) and a
# production container should not carry tools it never runs.
#
#   python -m venv .venv
#   .\.venv\Scripts\Activate.ps1
#   pip install -r requirements.txt -r requirements-dev.txt
#
# Versions are FLOORS (>=). After a successful install, freeze them:
#   pip freeze > requirements.lock.txt
# Benchmark numbers are only comparable across weeks if the tree is frozen.
# ---------------------------------------------------------------------------

# --- HTTP service ---------------------------------------------------------
fastapi>=0.115
uvicorn[standard]>=0.32
pydantic>=2.9
pydantic-settings>=2.6
python-dotenv>=1.0

# --- Talking to providers -------------------------------------------------
# Every provider adapter is built on httpx. The `openai` and `ollama` client
# packages were removed in A.2: providers are now reached over plain HTTP
# through switchboard/providers/, so those wrappers earned nothing.
httpx>=0.27

# --- Provider catalog -----------------------------------------------------

# --- Ledger ---------------------------------------------------------------

# Schema migrations. Required, not optional: without it, upgrading Switchboard

# PostgreSQL driver. SQLite is the default and needs nothing extra, but a team
# sharing one Switchboard needs a real database server.
psycopg[binary]>=3.2

# --- Live routing ---------------------------------------------------------
